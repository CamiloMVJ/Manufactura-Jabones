class ERPSystemTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.temp = tempfile.TemporaryDirectory(prefix="fridays-tests-")
        cls.db_path = Path(cls.temp.name) / "test.sqlite"
        cls.env = {**os.environ, "DATABASE_URL": "sqlite:///" + cls.db_path.as_posix(),
                   "SECRET_KEY": "test-only-secret-key-with-more-than-32-characters",
                   "APP_ENV": "development", "PYTHONDONTWRITEBYTECODE": "1"}
        # Solo el proceso de pruebas hereda estas variables, nunca se edita .env.
        os.environ.update({key: cls.env[key] for key in ("DATABASE_URL", "SECRET_KEY", "APP_ENV")})
        sys.path.insert(0, str(ROOT))
        from app.db.database import init_db, SessionLocal
        from app.models.erp_models import User, Client, Product
        from app.core.security import get_password_hash
        init_db()
        with SessionLocal() as db:
            hashed = get_password_hash("test-pass-123")
            for name, role, active in [("admin", "administrador", True), ("sales", "vendedor", True), ("driver", "repartidor", True), ("inactive", "administrador", False)]:
                db.add(User(username=name, email=name+"@example.test", hashed_password=hashed, role=role, is_active=active))
            db.add(Client(name="Cliente de prueba", email="cliente@example.test", zone="Centro"))
            for i in range(1, 24):
                db.add(Product(name=f"Producto {i:02}", price=10.25, stock=20 if i != 2 else 0, category="Granos"))
            db.commit()
        with socket.socket() as sock:
            sock.bind(("127.0.0.1", 0))
            cls.port = sock.getsockname()[1]
        cls.base = f"http://127.0.0.1:{cls.port}"
        cls.log = open(Path(cls.temp.name) / "server.log", "w", encoding="utf-8")
        cls.start_server()
        cls.tokens = {}
        for name in ("admin", "sales", "driver"):
            code, data = cls.request("POST", "/auth/login", {"username": name, "password": "test-pass-123"}, form=True)
            assert code == 200, data
            cls.tokens[name] = data["access_token"]

    @classmethod
    def start_server(cls):
        cls.stop_file = Path(cls.temp.name) / "stop-server"
        cls.stop_file.unlink(missing_ok=True)
        cls.env["ERP_TEST_STOP_FILE"] = str(cls.stop_file)
        cls.process = subprocess.Popen([sys.executable, str(ROOT / "tests" / "run_server.py"), str(cls.port)], cwd=ROOT, env=cls.env, stdout=cls.log, stderr=cls.log)
        for _ in range(100):
            try:
                with urlopen(cls.base+"/health", timeout=1) as response:
                    if response.status == 200:
                        return
            except OSError:
                if cls.process.poll() is not None:
                    break
                time.sleep(.1)
        raise RuntimeError("No inició el servidor de pruebas. Revisa el log: " + cls.temp.name)

    @classmethod
    def request(cls, method, path, data=None, user=None, form=False, raw=False):
        headers = {}
        if user:
            headers["Authorization"] = "Bearer " + cls.tokens[user]
        body = None
        if data is not None:
            body = (urlencode(data) if form else json.dumps(data)).encode()
            headers["Content-Type"] = "application/x-www-form-urlencoded" if form else "application/json"
        request = Request(cls.base+path, data=body, method=method, headers=headers)
        try:
            response = urlopen(request, timeout=15)
        except HTTPError as error:
            response = error
        with response:
            content = response.read()
            return response.status, content if raw else json.loads(content)

    @classmethod
    def tearDownClass(cls):
        cls.stop_server()
        cls.log.close()
        from app.db.database import engine
        engine.dispose()
        cls.temp.cleanup()

    @classmethod
    def stop_server(cls):
        cls.stop_file.touch()
        cls.process.wait(timeout=10)

    def test_development_secret_is_persistent_and_production_requires_configuration(self):
        from unittest.mock import patch
        from app.core import config
        config.get_secret_key.cache_clear()
        try:
            with patch.object(config, "BASE_DIR", Path(self.temp.name)), patch.object(config, "APP_ENV", "development"), patch.dict(os.environ, {"SECRET_KEY": ""}):
                first = config.get_secret_key()
                config.get_secret_key.cache_clear()
                self.assertEqual(config.get_secret_key(), first)
            config.get_secret_key.cache_clear()
            with patch.object(config, "APP_ENV", "production"), patch.dict(os.environ, {"SECRET_KEY": ""}):
                with self.assertRaises(RuntimeError):
                    config.get_secret_key()
        finally:
            config.get_secret_key.cache_clear()

    def test_order_failure_rolls_back_header_items_stock_and_log(self):
        before = self.request("GET", "/products/", user="admin")[1][0]["stock"]

    def test_duplicate_lines_and_status_workflow(self):
        code, order = self.request("POST", "/orders/", {"client_id":1,"items":[{"product_id":3,"quantity":1},{"product_id":3,"quantity":2}]}, "sales")
        self.assertEqual(code, 200, order)
        self.assertEqual(order["total_amount"], 30.75)
        detail = self.request("GET", f"/orders/{order['id']}", user="admin")[1]
        self.assertEqual(len(detail["items"]), 1)
        self.assertEqual(detail["items"][0]["quantity"], 3)
        route = f"/orders/{order['id']}/status"
        self.assertEqual(self.request("PATCH", route, {"status":"entregado"}, "admin")[0], 409)
        self.assertEqual(self.request("PATCH", route, {"status":"preparado"}, "driver")[0], 403)
        for status, user in [("preparado","sales"),("en_ruta","driver"),("entregado","driver")]:
            self.assertEqual(self.request("PATCH", route, {"status":status}, user)[0], 200)
        self.assertEqual(self.request("DELETE", "/products/3", user="admin")[0], 400)
        recent = self.request("GET", "/reports/recent-orders", user="admin")[1]
        self.assertEqual(next(o for o in recent if o["id"] == order["id"])["status"], "entregado")

    def test_concurrent_orders_do_not_oversell(self):
        self.request("PUT", "/products/4/stock?quantity=1", user="admin")
        data = {"client_id":1,"items":[{"product_id":4,"quantity":1}]}
        with ThreadPoolExecutor(max_workers=2) as executor:
            results = list(executor.map(lambda _: self.request("POST", "/orders/", data, "sales")[0], range(2)))
        self.assertEqual(sorted(results), [200,409])
        stock = next(p for p in self.request("GET", "/products/", user="admin")[1] if p["id"] == 4)["stock"]
        self.assertEqual(stock, 0)

    def test_finance_partial_payment_balance_and_idempotency(self):
        order_id = self.make_finance_order()
        data={"order_id":order_id,"amount":"7.10","method":"transferencia","reference":"TEST-PARTIAL","request_id":str(uuid4())}
        code, payment = self.request("POST", "/finance/payments",data,"sales")
        self.assertEqual(code,201,payment)
        repeated = self.request("POST", "/finance/payments",data,"sales")
        self.assertEqual(repeated[0],201,repeated)
        self.assertEqual(repeated[1]["id"],payment["id"])
        self.assertEqual(self.request("POST", "/finance/payments",{**data,"amount":"6.00"},"sales")[0],409)
        accounts = self.request("GET", "/finance/overview",user="admin")[1]["receivables"]
        account = next(r for r in accounts if r["id"] == order_id)
        self.assertEqual(Decimal(account["balance"]),Decimal("13.40"))
        self.assertEqual(Decimal(account["paid"]),Decimal("7.10"))
        self.assertEqual(self.request("POST", "/finance/payments",{**data,"request_id":str(uuid4()),"amount":"13.41"},"sales")[0],409)
        self.assertEqual(self.request("POST", "/finance/payments",{**data,"request_id":str(uuid4()),"amount":"13.40"},"sales")[0],201)
        account = next(r for r in self.request("GET", "/finance/overview",user="admin")[1]["receivables"] if r["id"] == order_id)
        self.assertEqual(Decimal(account["balance"]),Decimal("0"))

    def test_finance_concurrent_payments_cannot_exceed_order(self):
        order_id = self.make_finance_order()
        def pay(_):
            return self.request("POST", "/finance/payments",{"order_id":order_id,"amount":"15.00","method":"efectivo","request_id":str(uuid4())},"sales")[0]
        with ThreadPoolExecutor(max_workers=2) as executor:
            codes = list(executor.map(pay,range(2)))
        self.assertEqual(sorted(codes),[201,409])
        account = next(r for r in self.request("GET", "/finance/overview",user="admin")[1]["receivables"] if r["id"] == order_id)
        self.assertEqual(Decimal(account["balance"]),Decimal("5.50"))

    def test_finance_expense_idempotency_and_summary(self):
        before = self.request("GET", "/finance/overview",user="admin")[1]["summary"]
        data={"description":"Gasto de verificación", "category":"transporte", "amount":"4.35", "method":"efectivo", "request_id":str(uuid4())}
        code, expense = self.request("POST", "/finance/expenses",data,"admin")
        self.assertEqual(code,201,expense)
        repeated = self.request("POST", "/finance/expenses",data,"admin")
        self.assertEqual(repeated[1]["id"],expense["id"])
        self.assertEqual(self.request("POST", "/finance/expenses",{**data,"amount":"5.00"},"admin")[0],409)
        data = self.request("GET", "/finance/overview",user="admin")[1]
        self.assertEqual(Decimal(data["summary"]["expenses"])-Decimal(before["expenses"]),Decimal("4.35"))
        self.assertEqual(Decimal(data["summary"]["net_recorded"]),Decimal(data["summary"]["collected"])-Decimal(data["summary"]["expenses"]))
        self.assertEqual(len(data["trend"]),7)

    def test_websocket_authentication(self):
        import websockets
        async def check():
            async with websockets.connect(f"ws://127.0.0.1:{self.port}/ws/updates") as ws:
                await ws.send(self.tokens["admin"])
                self.assertEqual(json.loads(await ws.recv())["type"], "CONNECTED")
            async with websockets.connect(f"ws://127.0.0.1:{self.port}/ws/updates") as ws:
                await ws.send("invalid-token")
                with self.assertRaises(websockets.exceptions.ConnectionClosed):
                    await ws.recv()
                self.assertEqual(ws.close_code, 1008)
        asyncio.run(check())
