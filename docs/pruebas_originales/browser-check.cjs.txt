const sleep = (ms) => new Promise((r) => setTimeout(r,ms));
let ws,seq=0; const pending = new Map(), errors = [], results = [];
const command = (method,params={}) => new Promise((resolve,reject) => {
  const id = ++seq;
  const timer = setTimeout(() => {pending.delete(id);reject(new Error('CDP timeout: '+method));},15000);
  pending.set(id,{resolve:(v) => {clearTimeout(timer);resolve(v);},reject});
  ws.send(JSON.stringify({id,method,params}));
});
async function evaluate(expression) {
  const data = await command('Runtime.evaluate',{expression,awaitPromise:true,returnByValue:true});
  if (data.exceptionDetails) throw new Error(data.exceptionDetails.text+': '+data.exceptionDetails.exception?.description);
  return data.result.value;
}
async function until(expression,message) {
  for (let i=0;i<100;i++) { if (await evaluate(expression)) return; await sleep(100); }
  throw new Error('UI timeout: '+message+' at '+await evaluate('location.href'));
}
async function navigate(route) {
  await command('Page.navigate',{url:base+route});
  await until(`document.readyState === 'complete' && document.body?.dataset.page`, 'page loaded');
  if (route !== '/login') await until(`document.getElementById('last-updated')?.textContent.startsWith('Actualizado')`, 'data loaded');
}
async function click(selector) { await evaluate(`document.querySelector(${JSON.stringify(selector)}).click()`); }
async function value(selector,v) { await evaluate(`(() => {const n=document.querySelector(${JSON.stringify(selector)}); n.value=${JSON.stringify(v)}; n.dispatchEvent(new Event('input',{bubbles:true})); n.dispatchEvent(new Event('change',{bubbles:true}));})()`); }
async function screenshot(name) {
  const {data} = await command('Page.captureScreenshot',{format:'png',captureBeyondViewport:false});
  fs.writeFileSync(path.join(out,name+'.png'),Buffer.from(data,'base64'));
}
async function viewport(width,height) { await command('Emulation.setDeviceMetricsOverride',{width,height,deviceScaleFactor:1,mobile:width<700}); await sleep(150); }
async function noOverflow() {
  const dimensions = await evaluate('({viewport:innerWidth,body:document.body.scrollWidth,root:document.documentElement.scrollWidth})');
  if (dimensions.root > dimensions.viewport+1) console.error(await evaluate(`Array.from(document.querySelectorAll('body *')).filter(n=>n.getBoundingClientRect().right>innerWidth+1).map(n=>({tag:n.tagName,cls:n.className,id:n.id,right:n.getBoundingClientRect().right,width:n.getBoundingClientRect().width})).slice(0,20)`));
  assert(dimensions.root <= dimensions.viewport+1,JSON.stringify(dimensions));
}
